@VFS_OPENFILE
- Description: Opens an existing file and returns its file descriptor in x0.
- Param x1: The highslot containing the path string.
- Param x2: The length of the path string.
- Return x0: The file descriptor number.
SETVAL 0 3
SETVAL 1 x2
SETVAL 2 x1
SETVAL 3 0
SYSCALL
RET
@\

@VFS_CREATEFILE
- Description: Creates a file and returns its file descriptor in x0.
- Param x1: The highslot containing the path string.
- Param x2: The length of the path string.
- Param x3: The file type number.
- Return x0: The file descriptor number.
SETVAL 0 4
SETVAL 1 x2
SETVAL 2 x1
SETVAL 3 x3
SETVAL 4 0
SYSCALL
RET
@\

@VFS_WRITEFILECHAR
- Description: Writes a single character from [x1][x2] into an open file descriptor x3 at index x4.
- Param x1: The highslot where the character is stored.
- Param x2: The lowslot where the character is stored.
- Param x3: The file descriptor to write to.
- Param x4: The index in the file to write to.
SETVAL 0 5
SETVAL 1 x1
SETVAL 2 x2
SETVAL 3 x3
SETVAL 4 x4
SYSCALL
RET
@\

@VFS_WRITEFILESTR
- Description: Writes a full string from a highslot to an open file.
- Param x1: The highslot to read from.
- Param x2: How many characters to write.
- Param x3: The file descriptor to write to.
- Modifies x8: Left at the length count after completion.
SETVAL 5 x1
SETVAL 6 x2
SETVAL 7 x3
SETVAL 8 0
SETVAL 1 x5
SETVAL 2 x8
SETVAL 3 x7
SETVAL 4 x8
GO @VFS_WRITEFILECHAR
ADD 8 x8 1
JEQ x8 x6 +2
JMP -7
RET
@\

@VFS_READFILECHAR
- Description: Reads a single character from an open file descriptor x3 at index x4 into [x1][x2].
- Param x1: The highslot to store the character into.
- Param x2: The lowslot to store the character into.
- Param x3: The file descriptor to read from.
- Param x4: The index in the file to read from.
SETVAL 0 6
SETVAL 1 x1
SETVAL 2 x2
SETVAL 3 x3
SETVAL 4 x4
SYSCALL
RET
@\

@VFS_READFILESTR
- Description: Reads a full string from an open file descriptor into a highslot in LRAM.
- Param x1: The highslot to store the string into.
- Param x2: How many characters to read.
- Param x3: The file descriptor to read from.
- Modifies x8: Left at the length count after completion.
SETVAL 5 x1
SETVAL 6 x2
SETVAL 7 x3
SETVAL 8 0
SETVAL 1 x5
SETVAL 2 x8
SETVAL 3 x7
SETVAL 4 x8
GO @VFS_READFILECHAR
ADD 8 x8 1
JEQ x8 x6 +2
JMP -7
RET
@\

@VFS_CLOSEFILE
- Description: Closes an open file descriptor.
- Param x1: The file descriptor to close.
SETVAL 0 7
SETVAL 1 x1
SYSCALL
RET
@\
